# Chatterbox-TTS Service — pinned dependencies for reproducible installs.
#
# This file exists alongside pyproject.toml so a plain
# `pip install -r requirements.txt` (no build backend, no editable
# install) is always available as a fallback — mirrors the sibling
# WhisperSTTService convention. Keep both files in sync: if you add a
# dep to pyproject.toml, add it here too.

# MLX text-to-speech models (Chatterbox / Chatterbox-Turbo, and now
# Qwen3-TTS via qwen_engine.py), Apple Silicon / Metal only. Pinned
# >=0.4.5: first version confirmed to ship
# mlx_audio.tts.models.qwen3_tts — do not drop below this floor without
# re-verifying qwen3_tts is still importable.
mlx-audio>=0.4.5

# Pinned below 5.13: mlx-lm 0.31 (pulled in transitively by mlx-audio)
# breaks against transformers>=5.13. Do not bump past <5.13 without
# re-verifying mlx-lm compatibility first.
transformers>=5.5,<5.13

# Apple-silicon ML array framework backing mlx-audio.
mlx

# WebSocket server (binary + text frames, async, minimal footprint).
websockets>=13,<14

# Audio I/O + resampling.
soundfile
soxr

# Opus encoder for the gateway's Opus-everywhere audio pipeline.
opuslib

# Numeric.
numpy

# In-process CPU + RSS sampling.
psutil

# YAML config file loader.
pyyaml

# HuggingFace download helper — fetches/caches the MLX model weights.
huggingface_hub
