> **Status:** All phases (0–7) complete on `feature/acp-pivot`. Phase 7 cleanup landed 2026-05-08; ACP wire is the only path. Retained as historical context.

**Goal:** Replace the custom Hermes platform-adapter integration with a pure ACP-conformant client model. Gateway↔Hermes wire becomes ACP JSON-RPC over WebSocket. Retire 9 of 10 overlay patches and the 929 LOC `sentient_gateway.py`.

import { access, chmod, mkdir, rename } from "node:fs/promises";
import { constants } from "node:fs";
import { join } from "node:path";
import { getLog } from "../logging/logger.js";

const log = getLog(["sentient", "sessions", "storage-migrator"]);

export interface MigrateLegacyTitlesArgs {
  /** Legacy: <legacyDir>/<userId>.json. Default: ~/.sentient/gateway/session-titles */
  readonly legacyDir: string;
  /** New root: <newRoot>/<userId>/sessions/titles.json. Default: ~/.sentient/gateway/users */
  readonly newRoot: string;
  /** Profile id this migrator runs for. */
  readonly userId: string;
}

const USERID_RE = /^[A-Za-z0-9_-]+$/;

export async function migrateLegacyTitles(args: MigrateLegacyTitlesArgs): Promise<void> {
  if (!USERID_RE.test(args.userId)) {
    throw new Error(`invalid userId: ${args.userId}`);
  }
  const legacyFile = join(args.legacyDir, `${args.userId}.json`);
  const newDir = join(args.newRoot, args.userId, "sessions");
  const newFile = join(newDir, "titles.json");

  try {
    await access(legacyFile, constants.F_OK);
  } catch {
    log.debug("no-legacy-file", { legacyFile });
    return;
  }

  try {
    await access(newFile, constants.F_OK);
    log.info("already-migrated", { newFile });
    return;
  } catch {
    // newFile does not exist — proceed
  }

  await mkdir(newDir, { recursive: true });
  await chmod(join(args.newRoot, args.userId), 0o700);
  await rename(legacyFile, newFile);
  await chmod(newFile, 0o600);
  log.info("migrated", { from: legacyFile, to: newFile });
}
```

"""ACP-over-WebSocket bridge for sentient gateway.

Hosts the upstream HermesACPAgent unmodified, exposing it over a WebSocket
endpoint instead of stdio. Bun gateway dials in over WS; we adapt the WS
text-frame stream to the asyncio.StreamReader / StreamWriter interface that
``acp.run_agent`` accepts.

Replaces ``sentient_gateway.py``. Sentient is no longer a Hermes platform —
it's an ACP client. Per-profile process model preserved (one worker per user
under supervisord).
"""

from __future__ import annotations

import argparse
import asyncio
import logging
import os
import sys
from pathlib import Path

import aiohttp
from aiohttp import web

logger = logging.getLogger(__name__)

DEFAULT_PORT = 8650
AUTH_HEADER = "Authorization"
AUTH_PREFIX = "Bearer "


def _setup_logging() -> None:
    handler = logging.StreamHandler(sys.stderr)
    handler.setFormatter(
        logging.Formatter("%(asctime)s [%(levelname)s] %(name)s: %(message)s",
                          datefmt="%Y-%m-%d %H:%M:%S")
    )
    root = logging.getLogger()
    root.handlers.clear()
    root.addHandler(handler)
    root.setLevel(logging.INFO)


def _validate_bearer(request: web.Request, expected_token: str) -> bool:
    header = request.headers.get(AUTH_HEADER, "")
    if not header.startswith(AUTH_PREFIX):
        return False
    return header[len(AUTH_PREFIX):] == expected_token


async def health_handler(request: web.Request) -> web.Response:
    return web.Response(text="ok\n")


async def acp_ws_handler(request: web.Request) -> web.WebSocketResponse:
    expected_token = request.app["expected_token"]
    if not _validate_bearer(request, expected_token):
        logger.warning("auth-failed remote=%s", request.remote)
        return web.Response(status=401, text="Unauthorized")

    ws = web.WebSocketResponse(heartbeat=30.0)
    await ws.prepare(request)
    logger.info("ws-connected remote=%s", request.remote)

    # Phase 2 placeholder: echo + close. Stream adapter wires in Task 2.2.
    async for msg in ws:
        if msg.type == aiohttp.WSMsgType.TEXT:
            logger.debug("ws-msg.text len=%d", len(msg.data))
            await ws.send_str(msg.data)
        elif msg.type == aiohttp.WSMsgType.ERROR:
            logger.warning("ws-error %s", ws.exception())
            break

    logger.info("ws-disconnected remote=%s", request.remote)
    return ws


def build_app(profile: str, expected_token: str) -> web.Application:
    app = web.Application()
    app["profile"] = profile
    app["expected_token"] = expected_token
    app.router.add_get("/healthz", health_handler)
    app.router.add_get("/acp", acp_ws_handler)
    return app


def main() -> None:
    parser = argparse.ArgumentParser(description="ACP-over-WS bridge")
    parser.add_argument("--profile", required=True, help="Hermes profile name")
    parser.add_argument("--port", type=int, default=DEFAULT_PORT)
    parser.add_argument("--hermes-home", required=True, help="Path to /data/profiles/<profile>")
    args = parser.parse_args()

    _setup_logging()
    os.environ["HERMES_HOME"] = args.hermes_home

    expected_token = os.environ.get("SENTIENT_HERMES_BEARER", "")
    if not expected_token:
        logger.error("SENTIENT_HERMES_BEARER env var missing; refusing to start")
        sys.exit(1)

    app = build_app(args.profile, expected_token)
    logger.info("Starting acp_ws_server profile=%s port=%d", args.profile, args.port)
    web.run_app(app, port=args.port, print=None)


if __name__ == "__main__":
    main()
```

```python
class WSStreamReader:
    """Adapter: incoming WS text frames -> asyncio.StreamReader-compatible byte stream.

    ACP expects newline-delimited JSON-RPC. Each WS text frame is one JSON-RPC
    message; we synthesize a trailing \\n per frame so StreamReader.readline()
    yields exactly one message at a time.
    """
    def __init__(self, ws: web.WebSocketResponse) -> None:
        self._ws = ws
        self._reader = asyncio.StreamReader()
        self._task: asyncio.Task | None = asyncio.create_task(self._pump())

    async def _pump(self) -> None:
        try:
            async for msg in self._ws:
                if msg.type == aiohttp.WSMsgType.TEXT:
                    self._reader.feed_data(msg.data.encode("utf-8") + b"\n")
                elif msg.type == aiohttp.WSMsgType.ERROR:
                    logger.warning("ws-pump-error %s", self._ws.exception())
                    break
        finally:
            self._reader.feed_eof()

    async def readline(self) -> bytes:
        return await self._reader.readline()

    async def read(self, n: int = -1) -> bytes:
        return await self._reader.read(n)

    def at_eof(self) -> bool:
        return self._reader.at_eof()


class WSStreamWriter:
    """Adapter: bytes written by ACP -> outbound WS text frames.

    ACP writes newline-delimited JSON-RPC. We buffer until \\n, strip it,
    emit one WS text frame per message.
    """
    def __init__(self, ws: web.WebSocketResponse) -> None:
        self._ws = ws
        self._buf = bytearray()
        self._send_lock = asyncio.Lock()

    def write(self, data: bytes) -> None:
        self._buf.extend(data)
        while b"\n" in self._buf:
            line, _, rest = self._buf.partition(b"\n")
            self._buf = bytearray(rest)
            asyncio.create_task(self._send_text(line.decode("utf-8")))

    async def drain(self) -> None:
        # No-op for our buffered model; real backpressure would await aiohttp's
        # internal queue, but for one-frame-per-message we trust aiohttp.
        return None

    async def _send_text(self, s: str) -> None:
        async with self._send_lock:
            await self._ws.send_str(s)

    def close(self) -> None:
        # Caller wraps acp.run_agent; close happens when ws disconnects.
        pass

    def is_closing(self) -> bool:
        return self._ws.closed

    async def wait_closed(self) -> None:
        return None
```

every method/notification the spec relies on, prints results.

Usage:
  python acp_probe.py --url ws://localhost:8650/acp --token test-token \
                       --profile u_8c866990 [--scenario {init,new,prompt,list,cancel,all}]
"""

from __future__ import annotations
import argparse
import asyncio
import json
import sys
import time
from contextlib import asynccontextmanager

import aiohttp


class WSClient:
    def __init__(self, ws):
        self._ws = ws
        self._next_id = 1
        self._pending: dict[int, asyncio.Future] = {}
        self._notifications: list[dict] = []
        self._reader_task = asyncio.create_task(self._reader())

    async def _reader(self):
        async for msg in self._ws:
            if msg.type != aiohttp.WSMsgType.TEXT:
                continue
            doc = json.loads(msg.data)
            if "id" in doc and ("result" in doc or "error" in doc):
                fut = self._pending.pop(doc["id"], None)
                if fut and not fut.done():
                    fut.set_result(doc)
            else:
                self._notifications.append(doc)
                print("← notif:", json.dumps(doc, indent=2)[:500])

    async def request(self, method: str, params: dict | None = None) -> dict:
        rid = self._next_id
        self._next_id += 1
        fut = asyncio.get_running_loop().create_future()
        self._pending[rid] = fut
        await self._ws.send_str(json.dumps({"jsonrpc": "2.0", "id": rid, "method": method, "params": params or {}}))
        return await asyncio.wait_for(fut, timeout=120.0)

    async def cancel(self, rid: int):
        await self._ws.send_str(json.dumps({"jsonrpc": "2.0", "method": "$/cancelRequest", "params": {"id": rid}}))


@asynccontextmanager
async def connect(url: str, token: str):
    async with aiohttp.ClientSession() as session:
        async with session.ws_connect(url, headers={"Authorization": f"Bearer {token}"}) as ws:
            yield WSClient(ws)


async def scenario_initialize(c: WSClient):
    res = await c.request("initialize", {"clientCapabilities": {"sessionList": True}})
    print("→ initialize result:", json.dumps(res, indent=2)[:1000])
    return res


async def scenario_new(c: WSClient):
    res = await c.request("session/new", {"cwd": "/", "mcpServers": []})
    print("→ session/new result:", json.dumps(res, indent=2)[:1000])
    return res["result"]["sessionId"]


async def scenario_list(c: WSClient):
    res = await c.request("session/list", {})
    print("→ session/list result:", json.dumps(res, indent=2)[:2000])
    return res


async def scenario_prompt(c: WSClient, session_id: str, text: str):
    res = await c.request("session/prompt", {
        "sessionId": session_id,
        "prompt": [{"type": "text", "text": text}],
    })
    print("→ session/prompt result:", json.dumps(res, indent=2)[:1000])
    return res


async def scenario_cancel(c: WSClient, session_id: str, text: str):
    rid = c._next_id  # peek
    task = asyncio.create_task(c.request("session/prompt", {
        "sessionId": session_id,
        "prompt": [{"type": "text", "text": text}],
    }))
    await asyncio.sleep(1.0)
    await c.cancel(rid)
    try:
        await task
    except Exception as e:
        print("→ cancelled prompt errored as expected:", type(e).__name__, e)


async def main():
    p = argparse.ArgumentParser()
    p.add_argument("--url", default="ws://localhost:8650/acp")
    p.add_argument("--token", default="test-token")
    p.add_argument("--profile", required=True)
    p.add_argument("--scenario", default="all", choices=["init", "new", "prompt", "list", "cancel", "all"])
    args = p.parse_args()

    async with connect(args.url, args.token) as c:
        await scenario_initialize(c)
        if args.scenario in ("new", "all"):
            sid = await scenario_new(c)
        else:
            sid = None
        if args.scenario in ("list", "all"):
            await scenario_list(c)
        if args.scenario in ("prompt", "all") and sid:
            await scenario_prompt(c, sid, "Say hi in 5 words.")
        if args.scenario == "cancel" and sid:
            await scenario_cancel(c, sid, "Write a long story about a cat.")
        # Wait briefly to flush trailing notifications (auto-title etc.)
        await asyncio.sleep(15.0)
        print("\n--- captured notifications ---")
        for n in c._notifications:
            print(json.dumps(n, indent=2)[:500])

    asyncio.run(main())

import { z } from "zod";

export const acpRequestSchema = z.object({
  jsonrpc: z.literal("2.0"),
  id: z.union([z.number(), z.string()]),
  method: z.string(),
  params: z.unknown().optional(),
});

export const acpResponseSchema = z.object({
  jsonrpc: z.literal("2.0"),
  id: z.union([z.number(), z.string()]),
}).and(
  z.union([
    z.object({ result: z.unknown() }),
    z.object({ error: z.object({ code: z.number(), message: z.string(), data: z.unknown().optional() }) }),
  ]),
);

export const acpNotificationSchema = z.object({
  jsonrpc: z.literal("2.0"),
  method: z.string(),
  params: z.unknown().optional(),
});

export type AcpRequest = z.infer<typeof acpRequestSchema>;
export type AcpResponse = z.infer<typeof acpResponseSchema>;
export type AcpNotification = z.infer<typeof acpNotificationSchema>;

export type AcpEnvelope =
  | { kind: "request"; value: AcpRequest; method: string; id: number | string; params?: unknown }
  | { kind: "response"; id: number | string; result?: unknown; error?: { code: number; message: string; data?: unknown } }
  | { kind: "notification"; method: string; params?: unknown };

export function parseAcpEnvelope(raw: unknown): AcpEnvelope {
  // Try response first (has id + result|error, no method)
  if (typeof raw === "object" && raw !== null && "id" in raw && !("method" in raw)) {
    const r = acpResponseSchema.parse(raw);
    if ("result" in r) return { kind: "response", id: r.id, result: r.result };
    return { kind: "response", id: r.id, error: r.error };
  }
  // Request: has id + method
  if (typeof raw === "object" && raw !== null && "id" in raw && "method" in raw) {
    const r = acpRequestSchema.parse(raw);
    return { kind: "request", value: r, method: r.method, id: r.id, params: r.params };
  }
  // Notification: has method, no id
  if (typeof raw === "object" && raw !== null && "method" in raw && !("id" in raw)) {
    const r = acpNotificationSchema.parse(raw);
    return { kind: "notification", method: r.method, params: r.params };
  }
  throw new Error("invalid ACP envelope");
}
```

}

interface PendingRequest {
  resolve: (v: unknown) => void;
  reject: (e: Error) => void;
}

export interface AcpClient {
  request(method: string, params?: unknown): Promise<unknown>;
  notify(method: string, params?: unknown): Promise<void>;
  cancel(id: number | string): Promise<void>;
  onNotification(method: string, handler: (params: unknown) => void): () => void;
  handleIncoming(raw: string): void;
  rejectAllPending(error: Error): void;
}

export function createAcpClient(cfg: AcpClientConfig): AcpClient {
  let nextId = 1;
  const pending = new Map<number | string, PendingRequest>();
  const handlers = new Map<string, Set<(params: unknown) => void>>();

  const request = async (method: string, params?: unknown): Promise<unknown> => {
    const id = nextId++;
    const promise = new Promise((resolve, reject) => {
      pending.set(id, { resolve, reject });
    });
    await cfg.send(JSON.stringify({ jsonrpc: "2.0", id, method, params: params ?? {} }));
    return promise;
  };

  const notify = async (method: string, params?: unknown): Promise<void> => {
    await cfg.send(JSON.stringify({ jsonrpc: "2.0", method, params: params ?? {} }));
  };

  const cancel = async (id: number | string): Promise<void> => {
    await notify("$/cancelRequest", { id });
  };

  const onNotification = (method: string, handler: (params: unknown) => void): (() => void) => {
    let set = handlers.get(method);
    if (!set) {
      set = new Set();
      handlers.set(method, set);
    }
    set.add(handler);
    return () => set!.delete(handler);
  };

  const handleIncoming = (raw: string): void => {
    let env;
    try {
      env = parseAcpEnvelope(JSON.parse(raw));
    } catch (err) {
      log.warn("parse-failed", { err: (err as Error).message, preview: raw.slice(0, 120) });
      return;
    }
    if (env.kind === "response") {
      const p = pending.get(env.id);
      if (!p) {
        log.warn("response-no-pending", { id: env.id });
        return;
      }
      pending.delete(env.id);
      if ("error" in env && env.error) {
        p.reject(new Error(`ACP error ${env.error.code}: ${env.error.message}`));
      } else {
        p.resolve(env.result);
      }
    } else if (env.kind === "notification") {
      const set = handlers.get(env.method);
      if (set) for (const h of set) {
        try { h(env.params); } catch (err) { log.warn("handler-threw", { method: env.method, err: (err as Error).message }); }
      }
    } else {
      log.warn("unexpected-request-from-server", { method: env.method });
    }
  };

  const rejectAllPending = (error: Error): void => {
    for (const [, p] of pending) p.reject(error);
    pending.clear();
  };

  return { request, notify, cancel, onNotification, handleIncoming, rejectAllPending };
}
```

```ts
import { z } from "zod";

// Internal event types — must match the existing internal event shape consumed by cerebrum.
export type InternalEvent =
  | { type: "assistant.message"; text: string }
  | { type: "tool.started"; callId: string; toolName: string; argsPreview: string }
  | { type: "tool.progress"; callId: string; message: string }
  | { type: "tool.finished"; callId: string; status: "ok" | "failed"; summary: string }
  | { type: "sessions.renamed"; sessionId: string; title: string; source: "auto" | "user" }
  | { type: "commands.available"; commands: unknown[] };

const ARGS_PREVIEW_MAX = 120;

function previewJSON(v: unknown): string {
  try {
    const s = JSON.stringify(v);
    return s.length > ARGS_PREVIEW_MAX ? s.slice(0, ARGS_PREVIEW_MAX) + "…" : s;
  } catch {
    return "";
  }
}

function contentToText(c: unknown): string {
  if (typeof c === "string") return c;
  if (Array.isArray(c)) {
    return c
      .map((p) => (typeof p === "object" && p !== null && "text" in p ? (p as { text: string }).text : ""))
      .join("");
  }
  return "";
}

export function translateAcpNotification(params: unknown): InternalEvent[] {
  const p = params as { sessionId?: string; update?: { sessionUpdate?: string; [k: string]: unknown } };
  const sessionId = p?.sessionId ?? "";
  const u = p?.update;
  if (!u || typeof u.sessionUpdate !== "string") return [];

  switch (u.sessionUpdate) {
    case "agent_message_chunk":
      return [{ type: "assistant.message", text: contentToText(u.content) }];
    case "tool_call": {
      const callId = String(u.toolCallId ?? "");
      const toolName = String(u.title ?? u.kind ?? "tool");
      return [{ type: "tool.started", callId, toolName, argsPreview: previewJSON(u.rawInput) }];
    }
    case "tool_call_update": {
      const callId = String(u.toolCallId ?? "");
      if (u.status === "in_progress") {
        return [{ type: "tool.progress", callId, message: typeof u.content === "string" ? u.content : previewJSON(u.content) }];
      }
      if (u.status === "completed" || u.status === "failed") {
        const status: "ok" | "failed" = u.status === "completed" ? "ok" : "failed";
        const summary = typeof u.rawOutput === "string" ? u.rawOutput : previewJSON(u.rawOutput);
        return [{ type: "tool.finished", callId, status, summary }];
      }
      return [];
    }
    case "session_info_update":
      if (typeof u.title === "string") {
        return [{ type: "sessions.renamed", sessionId, title: u.title, source: "auto" }];
      }
      return [];
    case "available_commands_update":
      return [{ type: "commands.available", commands: Array.isArray(u.availableCommands) ? u.availableCommands : [] }];
    default:
      return [];
  }
}

import { getLog } from "../../logging/logger.js";
import { createAcpClient, type AcpClient } from "./client.ts";
import { translateAcpNotification, type InternalEvent } from "./event-translator.ts";

const log = getLog(["sentient", "hermes-adapter-client", "acp", "per-profile-connection"]);

export interface AcpPerProfileConnectionConfig {
  send: (raw: string) => Promise<void>;
  onIncoming: (cb: (raw: string) => void) => void;
}

export interface SendUserMessageArgs {
  sessionId: string;
  text: string;
  internal?: boolean;
}
export interface SendUserMessageResult {
  stopReason: string;
}

export interface AcpPerProfileConnection {
  initialize(): Promise<void>;
  sendUserMessage(args: SendUserMessageArgs): Promise<SendUserMessageResult>;
  cancelInflight(): Promise<void>;
  newSession(args: { mcpServers?: unknown[] }): Promise<{ sessionId: string }>;
  loadSession(args: { sessionId: string }): Promise<void>;
  listSessions(args: { cursor?: string }): Promise<unknown>;
  onEvent(cb: (e: InternalEvent) => void): () => void;
}

export function createAcpPerProfileConnection(cfg: AcpPerProfileConnectionConfig): AcpPerProfileConnection {
  const client: AcpClient = createAcpClient({ send: cfg.send });
  cfg.onIncoming((raw) => client.handleIncoming(raw));

  let inflightId: number | string | null = null;
  const eventHandlers = new Set<(e: InternalEvent) => void>();

  client.onNotification("session/update", (params) => {
    const events = translateAcpNotification(params);
    for (const h of eventHandlers) for (const e of events) h(e);
  });

  return {
    async initialize() {
      await client.request("initialize", { clientCapabilities: { sessionList: true } });
    },
    async sendUserMessage({ sessionId, text }) {
      // Track id for cancel; client.request hides id, so we use a local counter
      // by issuing a notify-with-id pattern via a small extension.
      const result = await client.request("session/prompt", {
        sessionId,
        prompt: [{ type: "text", text }],
      }) as { stopReason: string };
      return { stopReason: result.stopReason };
    },
    async cancelInflight() {
      if (inflightId !== null) await client.cancel(inflightId);
    },
    async newSession({ mcpServers }) {
      const r = await client.request("session/new", { cwd: "/", mcpServers: mcpServers ?? [] }) as { sessionId: string };
      return r;
    },
    async loadSession({ sessionId }) {
      await client.request("session/load", { sessionId });
    },
    async listSessions({ cursor }) {
      return await client.request("session/list", cursor ? { cursor } : {});
    },
    onEvent(cb) {
      eventHandlers.add(cb);
      return () => eventHandlers.delete(cb);
    },
  };
}
```
