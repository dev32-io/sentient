# Phase 7.5 — Gateway-Owned Hermes Pool Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Move hermes-* container lifecycle out of `docker-compose.yml` and into the gateway. The gateway owns the internal hermes↔gateway auth token (generated, persisted, never user-visible), creates/starts/stops hermes containers via the docker socket, and reconciles the running pool against `cfg.hermes.profiles` at boot and after admin add/remove. After this phase, the user's only touch points for multi-user infra are the admin UI and `~/.sentient/gateway/config/config.yaml`. No human-edited `.env` entries, no per-slot compose blocks, no secret files on disk that the user maintains.

**Architecture:** New `internal-secrets-store` (mirrors `secrets-store` pattern but for gateway-generated values, not user-provided ones). Extended `docker-control` adapter with `create / start / stop / remove / inspect` shelling out via `Bun.spawn` (same pattern as today's `restart`). New `hermes-pool-reconciler` module computes desired-vs-actual container set and converges. Boot wires reconciler after the existing boot-migration step. `UserProvisioner.createUser` calls the reconciler for the affected slot instead of going through `runApply` (which only knows how to restart, not create). The renderer's output destination changes from `<userId>/.generated/{config.yaml,SOUL.md}` to `<userId>/{config.yaml,SOUL.md}` so the bind-mounted hermes container can actually read the rendered config (today's apply orchestrator writes to a path hermes never sees — it's load-bearing). Hermes containers bind-mount `~/.sentient/gateway/data/<userId>/` as `/opt/data`, so per-user runtime data (MEMORY.md, response_store.db, conversations/) lives alongside the gateway-rendered config in one userId-keyed directory. The `<repo>/profiles/<slot>/` directories are retired and Kevin's existing data is migrated to his userId dir on first boot.

**Tech Stack:** Bun + TypeScript strict (gateway), `Bun.spawn` for docker CLI, Vitest, Zod. No new package deps.

**Branch:** continues on `feature/multi-user-phase7-webui-admin` (Phase 7.5 is a sub-phase, not a separate branch).

**Spec:** `docs/superpowers/specs/2026-04-24-multi-user-auth-and-settings-design.md` — §3.1 (storage layout — load-bearing for the renderer-output redirect), §3.2 (hermes containers always-on, Compose `multi-user` profile gating removed — this plan goes one step further and removes the services from compose entirely), §5.3 (gateway as secrets authority — extended to internal infra tokens), §5.4 (api_key_env injection at hermes container start — implemented properly via the new container spec builder).

**Parent plan:** `docs/superpowers/plans/2026-04-25-multi-user-phase7-webui-admin.md` — this plan unblocks the smoke-test step (Task 7.11) which assumed compose-defined hermes-bob/family worked.

```ts
// gateway/src/admin/internal-secrets-store.ts
import { existsSync, readFileSync } from "node:fs";
import { chmod, writeFile, mkdir } from "node:fs/promises";
import { join } from "node:path";
import { randomBytes } from "node:crypto";
import type { Result } from "@sentient/protocol";
import { getLog } from "../logging/logger.js";

const log = getLog(["sentient", "gateway", "admin", "internal-secrets"]);
const FILE_NAME = "internal-secrets.json";
const FILE_MODE = 0o600;
const TOKEN_BYTES = 32;
const SCHEMA_VERSION = 1;

export interface InternalSecrets {
  hermesAuthToken: string; // 64-char hex
}

interface Persisted extends InternalSecrets {
  schemaVersion: number;
}

export type InternalSecretsError =
  | { kind: "io-error"; reason: string }
  | { kind: "corrupt-file"; reason: string };

export interface InternalSecretsStore {
  /** Load from disk if present; otherwise generate + persist + return. */
  loadOrInit(): Promise<Result<InternalSecrets, InternalSecretsError>>;
  /** Synchronous read after loadOrInit. Throws if loadOrInit wasn't called. */
  getHermesAuthTokenSync(): string;
}

export function createInternalSecretsStore(dataDir: string): InternalSecretsStore {
  const path = join(dataDir, FILE_NAME);
  let cached: InternalSecrets | null = null;

  return {
    async loadOrInit() {
      if (cached) return { ok: true, value: cached };
      try {
        if (existsSync(path)) {
          const raw = readFileSync(path, "utf8");
          const parsed = JSON.parse(raw) as Persisted;
          if (typeof parsed.hermesAuthToken !== "string" || !/^[0-9a-f]{64}$/.test(parsed.hermesAuthToken)) {
            return { ok: false, error: { kind: "corrupt-file", reason: "invalid hermesAuthToken" } };
          }
          cached = { hermesAuthToken: parsed.hermesAuthToken };
          log.info("loadOrInit.loaded", { fingerprint: fingerprint(cached.hermesAuthToken) });
          return { ok: true, value: cached };
        }
        await mkdir(dataDir, { recursive: true });
        const fresh: InternalSecrets = { hermesAuthToken: randomBytes(TOKEN_BYTES).toString("hex") };
        const payload: Persisted = { ...fresh, schemaVersion: SCHEMA_VERSION };
        const tmp = `${path}.tmp`;
        await writeFile(tmp, JSON.stringify(payload, null, 2), { mode: FILE_MODE });
        await chmod(tmp, FILE_MODE);
        // rename is atomic
        const { rename } = await import("node:fs/promises");
        await rename(tmp, path);
        await chmod(path, FILE_MODE);
        cached = fresh;
        log.info("loadOrInit.generated", { fingerprint: fingerprint(fresh.hermesAuthToken) });
        return { ok: true, value: cached };
      } catch (e: unknown) {
        const reason = (e as Error).message;
        log.error("loadOrInit.failed", { reason });
        return { ok: false, error: { kind: "io-error", reason } };
      }
    },
    getHermesAuthTokenSync() {
      if (!cached) throw new Error("InternalSecretsStore: getHermesAuthTokenSync called before loadOrInit");
      return cached.hermesAuthToken;
    },
  };
}

/** First 4 + ellipsis + last 4 — for log lines. */
function fingerprint(token: string): string {
  return `${token.slice(0, 4)}…${token.slice(-4)}`;
}
```

```ts
  stop(containerName: string, timeoutMs: number): Promise<Result<void, DockerControlError>>;
```

```ts
async create(spec, timeoutMs) {
  const cmd = buildCreateArgs(spec);
  return runDocker(cmd, timeoutMs, opts);
},
async start(name, timeoutMs) {
  return runDocker(["docker", "start", name], timeoutMs, opts);
},
async stop(name, timeoutMs) {
  const tSec = Math.max(1, Math.floor(timeoutMs / 1000) - 1);
  return runDocker(["docker", "stop", "-t", String(tSec), name], timeoutMs, opts);
},
async remove(name, timeoutMs) {
  return runDocker(["docker", "rm", name], timeoutMs, opts);
},
async inspect(name, timeoutMs) {
  const out = await runDockerCapturingStdout(["docker", "inspect", name], timeoutMs, opts);
  if (!out.ok) {
    const stderr = out.error.kind === "non-zero-exit" ? out.error.stderr : "";
    if (stderr.toLowerCase().includes("no such object")) {
      return { ok: true, value: { exists: false, running: false, labels: {}, health: null } };
    }
    return out;
  }
  const parsed = JSON.parse(out.value)[0];
  return {
    ok: true,
    value: {
      exists: true,
      running: parsed.State?.Running === true,
      labels: parsed.Config?.Labels ?? {},
      health: parsed.State?.Health?.Status ?? null,
    },
  };
},
```

`buildCreateArgs(spec)` is a pure helper that produces the exact arg array the test expects. Order matters for the test — keep it stable: name → labels (alphabetical by key) → env (alphabetical by key) → mounts → networks → restart policy → image.

```ts
// gateway/src/admin/hermes-pool-reconciler.ts
import type { Result } from "@sentient/protocol";
import type { ContainerCreateSpec, ContainerInspect, DockerControl, HealthPoller } from "../infrastructure/docker-control.js";
import { getLog } from "../logging/logger.js";

const log = getLog(["sentient", "gateway", "admin", "hermes-pool"]);
const MANAGED_LABEL = "sentient.managed-by";
const MANAGED_VALUE = "gateway";

export interface DesiredSlot {
  slotKey: string;
  userId: string;
}

export interface PoolProfile {
  container_name: string;
  port: number;
  url: string;
  role: string;
}

export interface ReconcileInput {
  desired: DesiredSlot[];
  profiles: Record<string, PoolProfile>;
  /** Slots to inspect for orphan cleanup. Defaults to keys of `profiles`. */
  slotsToInspect?: string[];
  timeoutMs: number;
}

export interface ReconcilerDeps {
  dockerControl: Pick<DockerControl, "inspect" | "create" | "start" | "stop" | "remove" | "restart">;
  healthPoller: HealthPoller;
  buildSpec: (input: { slotKey: string; userId: string; profile: PoolProfile }) => ContainerCreateSpec;
}

export type ReconcileError = { kind: "step-failed"; step: string; reason: string };

export async function reconcilePool(input: ReconcileInput, deps: ReconcilerDeps): Promise<Result<void, ReconcileError>> {
  const desiredMap = new Map(input.desired.map((d) => [d.slotKey, d]));
  const slotsToInspect = input.slotsToInspect ?? Object.keys(input.profiles);

  // 1. Reconcile each desired slot up.
  for (const slot of input.desired) {
    const profile = input.profiles[slot.slotKey];
    if (!profile) {
      log.warn("desired-slot-not-in-profiles", { slotKey: slot.slotKey });
      continue;
    }
    const inspect = await deps.dockerControl.inspect(profile.container_name, input.timeoutMs);
    if (!inspect.ok) return { ok: false, error: { kind: "step-failed", step: "inspect", reason: inspect.error.kind } };
    const state = inspect.value;

    if (!state.exists) {
      log.info("create.start", { slotKey: slot.slotKey });
      const spec = deps.buildSpec({ slotKey: slot.slotKey, userId: slot.userId, profile });
      const created = await deps.dockerControl.create(spec, input.timeoutMs);
      if (!created.ok) return { ok: false, error: { kind: "step-failed", step: "create", reason: created.error.kind } };
      const started = await deps.dockerControl.start(profile.container_name, input.timeoutMs);
      if (!started.ok) return { ok: false, error: { kind: "step-failed", step: "start", reason: started.error.kind } };
      continue;
    }
    if (!isManagedByGateway(state)) {
      log.warn("skipped-foreign-container", { containerName: profile.container_name });
      continue;
    }
    if (!state.running) {
      log.info("start", { slotKey: slot.slotKey });
      const started = await deps.dockerControl.start(profile.container_name, input.timeoutMs);
      if (!started.ok) return { ok: false, error: { kind: "step-failed", step: "start", reason: started.error.kind } };
    }
    // running + managed → leave alone
  }

  // 2. Reconcile orphan containers (managed, in pool, but no desired slot)
  for (const slotKey of slotsToInspect) {
    if (desiredMap.has(slotKey)) continue;
    const profile = input.profiles[slotKey];
    if (!profile) continue;
    const inspect = await deps.dockerControl.inspect(profile.container_name, input.timeoutMs);
    if (!inspect.ok) continue; // best-effort cleanup; don't fail boot on inspect glitches
    const state = inspect.value;
    if (!state.exists || !isManagedByGateway(state)) continue;

    log.info("orphan.stop-and-remove", { slotKey, containerName: profile.container_name });
    const stopped = await deps.dockerControl.stop(profile.container_name, input.timeoutMs);
    if (!stopped.ok) {
      log.warn("orphan.stop-failed", { slotKey, reason: stopped.error.kind });
      // proceed to remove anyway
    }
    const removed = await deps.dockerControl.remove(profile.container_name, input.timeoutMs);
    if (!removed.ok) {
      log.warn("orphan.remove-failed", { slotKey, reason: removed.error.kind });
    }
  }

  return { ok: true, value: undefined };
}

function isManagedByGateway(state: ContainerInspect): boolean {
  return state.labels[MANAGED_LABEL] === MANAGED_VALUE;
}
```

```ts
// gateway/src/admin/profile-dir-migration.ts
import { existsSync, statSync } from "node:fs";
import { readdir, rename, rm, mkdir } from "node:fs/promises";
import { join } from "node:path";
import type { Result } from "@sentient/protocol";
import { getLog } from "../logging/logger.js";

const log = getLog(["sentient", "gateway", "admin", "profile-dir-migration"]);

export interface MigrateInput {
  projectProfilesDir: string; // <repo>/profiles
  gatewayDataDir: string; // ~/.sentient/gateway/data
  bindings: { userId: string; slotKey: string }[];
}

export interface MigrationResult {
  moved: { slotKey: string; userId: string; fileCount: number }[];
}

export type MigrationError = { kind: "io-error"; reason: string };

export async function migrateProfileDirs(input: MigrateInput): Promise<Result<MigrationResult, MigrationError>> {
  if (!existsSync(input.projectProfilesDir)) {
    log.info("noop.no-source", { projectProfilesDir: input.projectProfilesDir });
    return { ok: true, value: { moved: [] } };
  }
  const moved: MigrationResult["moved"] = [];
  for (const { slotKey, userId } of input.bindings) {
    const src = join(input.projectProfilesDir, slotKey);
    const dst = join(input.gatewayDataDir, userId);
    if (!existsSync(src) || !statSync(src).isDirectory()) continue;
    await mkdir(dst, { recursive: true });
    let fileCount = 0;
    try {
      const entries = await readdir(src, { withFileTypes: true });
      for (const entry of entries) {
        const from = join(src, entry.name);
        const to = join(dst, entry.name);
        if (existsSync(to)) {
          log.warn("skip-existing", { slotKey, userId, file: entry.name });
          continue;
        }
        await rename(from, to);
        fileCount++;
      }
      // Remove the now-empty (or near-empty) source dir
      const remaining = await readdir(src);
      if (remaining.length === 0) await rm(src, { recursive: true, force: true });
      log.info("migrated", { slotKey, userId, fileCount });
      moved.push({ slotKey, userId, fileCount });
    } catch (e: unknown) {
      const reason = (e as Error).message;
```
